//Ring Based Program Execution
#include <mpi.h>     // Contains MPI function declarations
#include <stdio.h>   // Standard input-output

int main()
{
    int np;   // number of processes
    int pid;  // process ID

    MPI_Init(NULL, NULL);
    MPI_Comm_size(MPI_COMM_WORLD, &np);
    MPI_Comm_rank(MPI_COMM_WORLD, &pid);

    MPI_Status sta;

    if (pid == 0)
    {
        int Send_msg = 5;

        // Send message to next process
        MPI_Send(&Send_msg, 1, MPI_INT, pid + 1, 50, MPI_COMM_WORLD);

        printf("Process %d sent message %d to process %d\n", pid, Send_msg, pid + 1);

        // Receive message from last process
        MPI_Recv(&Send_msg, 1, MPI_INT, np - 1, 50, MPI_COMM_WORLD, &sta);

        printf("Process %d received message %d from process %d\n", pid, Send_msg, np - 1);
    }
    else if (pid != np - 1)
    {
        int Recv_msg;

        // Receive from previous process
        MPI_Recv(&Recv_msg, 1, MPI_INT, pid - 1, 50, MPI_COMM_WORLD, &sta);

        printf("Process %d received message %d from process %d\n", pid, Recv_msg, pid - 1);

        // Send to next process
        MPI_Send(&Recv_msg, 1, MPI_INT, pid + 1, 50, MPI_COMM_WORLD);

        printf("Process %d sent message %d to process %d\n", pid, Recv_msg, pid + 1);
    }
    else
    {
        int Recv_msg;

        // Last process receives from previous
        MPI_Recv(&Recv_msg, 1, MPI_INT, pid - 1, 50, MPI_COMM_WORLD, &sta);

        printf("Process %d received message %d from process %d\n", pid, Recv_msg, pid - 1);

        // Send back to process 0 (closing the ring)
        MPI_Send(&Recv_msg, 1, MPI_INT, 0, 50, MPI_COMM_WORLD);

        printf("Process %d sent message %d to process %d\n", pid, Recv_msg, 0);
    }

    MPI_Finalize();
    return 0;
}




//Ring Topology General Program Execution
#include <mpi.h>
#include <stdio.h>

int main()
{
	int pid, np;
	int msg = 5;
	MPI_Status status;

	MPI_Init(NULL, NULL);
	MPI_Comm_rank(MPI_COMM_WORLD, &pid);
	MPI_Comm_size(MPI_COMM_WORLD, &np);

	int next = (pid + 1) % np;
	int prev = (pid - 1 + np) % np;

	if (pid == 0)
	{
		MPI_Send(&msg, 1, MPI_INT, next, 50, MPI_COMM_WORLD);
		printf("Process %d sent %d to %d\n", pid, msg, next);

		MPI_Recv(&msg, 1, MPI_INT, prev, 50, MPI_COMM_WORLD, &status);
		printf("Process %d received %d from %d\n", pid, msg, prev);
	}
	else
	{
		MPI_Recv(&msg, 1, MPI_INT, prev, 50, MPI_COMM_WORLD, &status);
		printf("Process %d received %d from %d\n", pid, msg, prev);

		MPI_Send(&msg, 1, MPI_INT, next, 50, MPI_COMM_WORLD);
		printf("Process %d sent %d to %d\n", pid, msg, next);
	}

	MPI_Finalize();
	return 0;
}

		
//Tree Topology 	
#include <mpi.h>
#include <stdio.h>

int main()
{
    int pid, np;
    MPI_Status status;

    MPI_Init(NULL, NULL);
    MPI_Comm_rank(MPI_COMM_WORLD, &pid);
    MPI_Comm_size(MPI_COMM_WORLD, &np);

    int arr[6] = { 5,10,15,20,25,30 };
    int value, recv_val, sum;

    if (pid == 0)
    {
        for (int i = 1;i <= 5;i++)
        {
            MPI_Send(&arr[i], 1, MPI_INT, i, 0, MPI_COMM_WORLD);
        }

        int sum1, sum2;

        MPI_Recv(&sum1, 1, MPI_INT, 1, 0, MPI_COMM_WORLD, &status);
        MPI_Recv(&sum2, 1, MPI_INT, 3, 0, MPI_COMM_WORLD, &status);

        int total = arr[0] + sum1 + sum2;

        printf("Final Total Sum = %d\n", total);
    }

    else if (pid == 1)
    {
        MPI_Recv(&value, 1, MPI_INT, 0, 0, MPI_COMM_WORLD, &status);

        MPI_Recv(&recv_val, 1, MPI_INT, 2, 0, MPI_COMM_WORLD, &status);

        sum = value + recv_val;

        MPI_Send(&sum, 1, MPI_INT, 0, 0, MPI_COMM_WORLD);
    }

    else if (pid == 2)
    {
        MPI_Recv(&value, 1, MPI_INT, 0, 0, MPI_COMM_WORLD, &status);

        MPI_Send(&value, 1, MPI_INT, 1, 0, MPI_COMM_WORLD);
    }

    else if (pid == 3)
    {
        MPI_Recv(&value, 1, MPI_INT, 0, 0, MPI_COMM_WORLD, &status);

        MPI_Recv(&recv_val, 1, MPI_INT, 4, 0, MPI_COMM_WORLD, &status);

        sum = value + recv_val;

        MPI_Send(&sum, 1, MPI_INT, 0, 0, MPI_COMM_WORLD);
    }

    else if (pid == 4)
    {
        MPI_Recv(&value, 1, MPI_INT, 0, 0, MPI_COMM_WORLD, &status);

        MPI_Recv(&recv_val, 1, MPI_INT, 5, 0, MPI_COMM_WORLD, &status);

        sum = value + recv_val;

        MPI_Send(&sum, 1, MPI_INT, 3, 0, MPI_COMM_WORLD);
    }

    else if (pid == 5)
    {
        MPI_Recv(&value, 1, MPI_INT, 0, 0, MPI_COMM_WORLD, &status);

        MPI_Send(&value, 1, MPI_INT, 4, 0, MPI_COMM_WORLD);
    }

    MPI_Finalize();
    return 0;
}
